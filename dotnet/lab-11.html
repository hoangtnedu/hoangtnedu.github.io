5. Bài tập có hướng dẫn và code mẫu

1. Tạo Stored Procedure thêm sinh viên

Yêu cầu: Tạo sp_SinhVien_Insert có kiểm tra trùng mã và trả mã lỗi phù hợp. Gợi ý triển khai: Dùng THROW để truyền lỗi nghiệp vụ về ứng dụng.

  CREATE OR ALTER PROCEDURE sp_SinhVien_Insert
    @MaSV VARCHAR(15), @HoTen NVARCHAR(100), @NgaySinh DATE,
    @GioiTinh BIT, @MaKhoa VARCHAR(10), @DiemTB DECIMAL(4,2),
    @Email VARCHAR(100) = NULL
  AS
  BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM SinhVien WHERE MaSV=@MaSV)
      THROW 50001, N'Mã sinh viên đã tồn tại.', 1;

    INSERT INTO SinhVien(MaSV,HoTen,NgaySinh,GioiTinh,MaKhoa,DiemTB,Email)
    VALUES(@MaSV,@HoTen,@NgaySinh,@GioiTinh,@MaKhoa,@DiemTB,@Email);
  END;

GO

2. Stored Procedure cập nhật và xóa

Yêu cầu: Tạo sp_SinhVien_Update, sp_SinhVien_Delete; báo lỗi khi mã không tồn tại. Gợi ý triển khai: Không thay đổi khóa chính trong thủ tục cập nhật.

CREATE OR ALTER PROCEDURE sp_SinhVien_Update
  @MaSV VARCHAR(15), @HoTen NVARCHAR(100), @NgaySinh DATE,
  @GioiTinh BIT, @MaKhoa VARCHAR(10), @DiemTB DECIMAL(4,2),
  @Email VARCHAR(100) = NULL
AS
BEGIN
  UPDATE SinhVien SET HoTen=@HoTen, NgaySinh=@NgaySinh,
    GioiTinh=@GioiTinh, MaKhoa=@MaKhoa, DiemTB=@DiemTB, Email=@Email
  WHERE MaSV=@MaSV;
  IF @@ROWCOUNT=0 THROW 50002, N'Không tìm thấy sinh viên.', 1;
END;
GO

CREATE OR ALTER PROCEDURE sp_SinhVien_Delete @MaSV VARCHAR(15)
AS
BEGIN
  DELETE FROM SinhVien WHERE MaSV=@MaSV;
  IF @@ROWCOUNT=0 THROW 50002, N'Không tìm thấy sinh viên.', 1;
END;
GO

3. Tầng DAL gọi Stored Procedure

Yêu cầu: Viết Insert(SinhVienDTO dto) và trả số dòng ảnh hưởng. Gợi ý triển khai: Dùng CommandType.StoredProcedure và using cho tài nguyên.

public int Insert(SinhVienDTO dto)
{
  using (SqlConnection conn = DbConnectionFactory.Create())
  using (SqlCommand cmd = new SqlCommand("sp_SinhVien_Insert", conn))
  {
    cmd.CommandType = CommandType.StoredProcedure;
    cmd.Parameters.Add("@MaSV", SqlDbType.VarChar, 15).Value = dto.MaSV;
    cmd.Parameters.Add("@HoTen", SqlDbType.NVarChar, 100).Value = dto.HoTen;
    cmd.Parameters.Add("@NgaySinh", SqlDbType.Date).Value = dto.NgaySinh;
    cmd.Parameters.Add("@GioiTinh", SqlDbType.Bit).Value = dto.GioiTinh;
    cmd.Parameters.Add("@MaKhoa", SqlDbType.VarChar, 10).Value = dto.MaKhoa;
    cmd.Parameters.Add("@DiemTB", SqlDbType.Decimal).Value = dto.DiemTB;
    cmd.Parameters.Add("@Email", SqlDbType.VarChar, 100).Value =
       string.IsNullOrWhiteSpace(dto.Email) ? (object)DBNull.Value : dto.Email;
    conn.Open();
    return cmd.ExecuteNonQuery();
  }
}

Bài 2. Tạo trang chủ
Khi truy cập /, ứng dụng hiển thị trang chủ và nhận dữ liệu từ Controller.
HomeController.java
 package io.github.hoangtnedu.java.lab11.controller;
 import org.springframework.stereotype.Controller;
 import org.springframework.ui.Model;
 import org.springframework.web.bind.annotation.GetMapping;
 @Controller
 public class HomeController {
     @GetMapping("/")
     public String index(Model model) {
         model.addAttribute("title", "Hệ thống quản lý sinh viên");
         model.addAttribute("message", "Chào mừng đến với Spring Boot");
         return "index";
     }
 }


templates/index.html
 <!DOCTYPE html>
 <html xmlns:th="http://www.thymeleaf.org">
 <head>
     <meta charset="UTF-8">
     <title th:text="${title}">Trang chủ</title>
 </head>
 <body>
     <h1 th:text="${message}"></h1>
     <a th:href="@{/students}">Danh sách sinh viên</a>
 </body>
 </html>

Bài 4. Hiển thị danh sách sinh viên
Tạo URL /students để hiển thị danh sách sinh viên mẫu bằng Thymeleaf.
StudentController.java
 package io.github.hoangtnedu.java.lab11.controller;
 import org.springframework.stereotype.Controller;
 import org.springframework.ui.Model;
 import org.springframework.web.bind.annotation.GetMapping;
 import io.github.hoangtnedu.java.lab11.model.Student;
 import java.util.List;
 @Controller
 public class StudentController {
     @GetMapping("/students")
     public String listStudents(Model model) {
         List<Student> students = List.of(
             new Student("SV001", "Nguyễn Văn An", "an@example.com", "JAVA01"),
             new Student("SV002", "Trần Thị Bình", "binh@example.com", "JAVA01"),
             new Student("SV003", "Lê Văn Cường", "cuong@example.com", "JAVA01")
         );
         model.addAttribute("students", students);
         return "students";
     }
 }


templates/students.html
 <table border="1">
     <tr>
         <th>Mã SV</th>
         <th>Họ tên</th>
         <th>Email</th>
         <th>Lớp</th>
     </tr>
     <tr th:each="s : ${students}">
         <td th:text="${s.studentCode}"></td>
         <td th:text="${s.fullName}"></td>
         <td th:text="${s.email}"></td>
         <td th:text="${s.className}"></td>
     </tr>
 </table>

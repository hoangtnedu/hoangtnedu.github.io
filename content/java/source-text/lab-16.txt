5.2. Trạng thái và quy tắc chuyển trạng thái
Mã trạng thái dùng thống nhất trong cả ba ứng dụng
   public enum OrderStatus {
       PENDING,       // Khách hàng vừa đặt
       PROCESSING,    // Kho đã tiếp nhận và trừ tồn kho
       READY,         // Đã đóng gói, sẵn sàng giao

Script tạo lược đồ MySQL dùng chung
   CREATE DATABASE java_integrated_lab
   CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
   USE java_integrated_lab;

   CREATE TABLE users (
       id BIGINT PRIMARY KEY AUTO_INCREMENT,
       username VARCHAR(50) NOT NULL UNIQUE,
       password_hash VARCHAR(255) NOT NULL,
       full_name VARCHAR(100) NOT NULL,
       role VARCHAR(20) NOT NULL,
       enabled BOOLEAN NOT NULL DEFAULT TRUE,
       created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
   );

   CREATE TABLE products (
       id BIGINT PRIMARY KEY AUTO_INCREMENT,
       code VARCHAR(30) NOT NULL UNIQUE,
       name VARCHAR(150) NOT NULL,
       price DECIMAL(15,2) NOT NULL,
       stock INT NOT NULL DEFAULT 0,
       active BOOLEAN NOT NULL DEFAULT TRUE,
       version INT NOT NULL DEFAULT 0
   );

   CREATE TABLE orders (
       id BIGINT PRIMARY KEY AUTO_INCREMENT,
       customer_id BIGINT NOT NULL,
       created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
       total_amount DECIMAL(15,2) NOT NULL DEFAULT 0,
       status VARCHAR(20) NOT NULL,
       note VARCHAR(500),
       version INT NOT NULL DEFAULT 0,
       CONSTRAINT fk_orders_customer FOREIGN KEY (customer_id) REFERENCES users(id)
   );

   CREATE TABLE order_items (
       id BIGINT PRIMARY KEY AUTO_INCREMENT,
       order_id BIGINT NOT NULL,
       product_id BIGINT NOT NULL,
       quantity INT NOT NULL,
       unit_price DECIMAL(15,2) NOT NULL,
       subtotal DECIMAL(15,2) NOT NULL,
       CONSTRAINT fk_items_order FOREIGN KEY (order_id) REFERENCES orders(id),
       CONSTRAINT fk_items_product FOREIGN KEY (product_id) REFERENCES products(id)
   );

   CREATE TABLE order_status_history (
       id BIGINT PRIMARY KEY AUTO_INCREMENT,
       order_id BIGINT NOT NULL,
       old_status VARCHAR(20),
       new_status VARCHAR(20) NOT NULL,
       changed_by BIGINT NOT NULL,
       changed_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
       platform VARCHAR(30) NOT NULL,
       note VARCHAR(500),
       CONSTRAINT fk_history_order FOREIGN KEY (order_id) REFERENCES orders(id),
       CONSTRAINT fk_history_user FOREIGN KEY (changed_by) REFERENCES users(id)
   );

   CREATE INDEX idx_orders_status ON orders(status);
   CREATE INDEX idx_orders_customer ON orders(customer_id);
   CREATE INDEX idx_history_order ON order_status_history(order_id);

Entity Order gợi ý
    @Entity
    @Table(name = "orders")
    public class Order {
        @Id
        @GeneratedValue(strategy = GenerationType.IDENTITY)
        private Long id;

        @Column(name = "customer_id", nullable = false)
        private Long customerId;

        @Column(name = "created_at", nullable = false)
        private LocalDateTime createdAt;

        @Column(name = "total_amount", nullable = false)
        private BigDecimal totalAmount;

        @Enumerated(EnumType.STRING)
        @Column(nullable = false)
        private OrderStatus status;

        @Version
        private Integer version;

        @OneToMany(mappedBy = "order", cascade = CascadeType.ALL,
                   orphanRemoval = true)
        private List<OrderItem> items = new ArrayList<>();

        @PrePersist
        public void prePersist() {
            createdAt = LocalDateTime.now();
            status = OrderStatus.PENDING;
        }
    }


Service tạo đơn hàng gợi ý
    @Stateless
    public class OrderService {
        @PersistenceContext(unitName = "JavaIntegratedPU")
        private EntityManager entityManager;

        public Long createOrder(Long customerId,
                                List<CartItem> cartItems,
                                String note) {
            if (cartItems == null || cartItems.isEmpty()) {
                throw new IllegalArgumentException("Giỏ hàng đang trống.");
            }

            Order order = new Order();
            order.setCustomerId(customerId);
            order.setNote(note);
            BigDecimal total = BigDecimal.ZERO;

            for (CartItem cartItem : cartItems) {
                Product product = entityManager.find(
                    Product.class, cartItem.getProductId());
                if (product == null || !product.isActive()) {
                    throw new IllegalArgumentException("Sản phẩm không hợp lệ.");
                }

                OrderItem item = new OrderItem();
                item.setOrder(order);
                item.setProduct(product);
                item.setQuantity(cartItem.getQuantity());
                item.setUnitPrice(product.getPrice());
                item.setSubtotal(product.getPrice().multiply(
                    BigDecimal.valueOf(cartItem.getQuantity())));
                order.getItems().add(item);
                total = total.add(item.getSubtotal());
            }

            order.setTotalAmount(total);
            entityManager.persist(order);
            saveHistory(order.getId(), null, OrderStatus.PENDING,
                        customerId, "JAKARTA_EE", "Khách tạo đơn");
            return order.getId();
        }
    }

DAO lấy danh sách đơn theo trạng thái
    public List<Order> findOrdersByStatus(String status)
            throws SQLException {
        String sql = """
            SELECT o.id, o.customer_id, o.created_at,
                   o.total_amount, o.status,
                   u.full_name AS customer_name
            FROM orders o
            JOIN users u ON u.id = o.customer_id
            WHERE o.status = ?
            ORDER BY o.created_at ASC
            """;

        List<Order> orders = new ArrayList<>();
        try (Connection connection = DatabaseConnection.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {
            statement.setString(1, status);
            try (ResultSet rs = statement.executeQuery()) {
                while (rs.next()) {
                    Order order = new Order();
                    order.setId(rs.getLong("id"));
                    order.setCustomerName(rs.getString("customer_name"));
                    order.setTotalAmount(rs.getBigDecimal("total_amount"));
                    order.setStatus(rs.getString("status"));
                    orders.add(order);
                }
            }
        }
        return orders;
    }


Service tiếp nhận đơn - transaction và kiểm soát xung đột
    public void acceptOrder(Long orderId, Long employeeId)
            throws SQLException {
        try (Connection connection = DatabaseConnection.getConnection()) {
            connection.setAutoCommit(false);
            try {
                lockAndValidatePendingOrder(connection, orderId);
                List<OrderItem> items = loadItems(connection, orderId);

               for (OrderItem item : items) {
                   int affected = decreaseStockIfAvailable(
                       connection, item.getProductId(), item.getQuantity());
                   if (affected == 0) {
                       throw new IllegalStateException("Không đủ tồn kho.");
                   }
               }

               int updated = updateStatusIfCurrent(
                   connection, orderId, "PENDING", "PROCESSING");
               if (updated == 0) {
                   throw new IllegalStateException("Đơn đã được xử lý.");
               }

                insertHistory(connection, orderId, "PENDING", "PROCESSING",
                              employeeId, "JAVA_SWING",
                              "Nhân viên kho tiếp nhận đơn");
                connection.commit();
            } catch (Exception ex) {
                connection.rollback();
                throw ex;
            } finally {
                connection.setAutoCommit(true);
            }
        }
    }

Service phê duyệt giao hàng
   @Service
   public class ManagementOrderService {
       private final OrderRepository orderRepository;
       private final OrderStatusHistoryRepository historyRepository;

        @Transactional
        public void approveShipping(Long orderId, Long managerId) {
            Order order = orderRepository.findById(orderId)
                .orElseThrow(() -> new IllegalArgumentException(
                    "Không tìm thấy đơn hàng."));

            if (order.getStatus() != OrderStatus.READY) {
                throw new IllegalStateException(
                    "Chỉ đơn READY mới được phê duyệt giao hàng.");
            }

            order.setStatus(OrderStatus.SHIPPING);
            orderRepository.save(order);

            OrderStatusHistory history = new OrderStatusHistory();
            history.setOrderId(order.getId());
            history.setOldStatus(OrderStatus.READY);
            history.setNewStatus(OrderStatus.SHIPPING);
            history.setChangedBy(managerId);
            history.setPlatform("SPRING_BOOT");
            history.setNote("Quản lý phê duyệt giao hàng");
            historyRepository.save(history);
        }
   }

10. Yêu cầu kỹ thuật bắt buộc
10.1. Yêu cầu chung
•   Cả ba ứng dụng dùng Maven và cùng kết nối một cơ sở dữ liệu MySQL.
•   Không cho phép người dùng nhập trực tiếp trạng thái đơn hàng.
•   Không đặt toàn bộ logic trong JFrame, Servlet hoặc Controller.
•   Các thao tác cập nhật nhiều bảng phải dùng transaction và rollback khi lỗi.
•   JDBC dùng PreparedStatement, try-with-resources và xử lý ngoại lệ phù hợp.
•   Mã nguồn tổ chức package rõ ràng; thông tin kết nối không hard-code trong sản phẩm hoàn chỉnh.

10.2. Java Swing
•   Áp dụng MVC; SQL chỉ nằm trong DAO/Repository.
•   Không chạy truy vấn dài trên Event Dispatch Thread; khuyến khích SwingWorker.
•   Làm mới JTable sau cập nhật và vô hiệu nút thao tác không hợp lệ.

10.3. Jakarta EE
•   Servlet/JSF chỉ tiếp nhận và điều hướng request; Service xử lý nghiệp vụ.
•   JPA Entity ánh xạ đúng schema; chỉ hiển thị đơn của khách hàng hiện tại.
•   Quản lý session và kiểm tra quyền trước khi thực hiện chức năng.

10.4. Spring Boot
•   Tách Controller - Service - Repository.
•   Sử dụng @Transactional cho nghiệp vụ cập nhật.
•   Spring Security phân quyền theo vai trò; không trả mật khẩu ra giao diện.


•     Kiểm tra trạng thái/version trước khi cập nhật.
